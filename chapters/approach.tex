\subsection{Primary Path}

\begin{itemize}[leftmargin=1.8em, itemsep=3pt]
  \item \textbf{Engine:} NVIDIA Triton Inference Server
  \item \textbf{การประมวลผล:} เร่งด้วย GPU (gRPC + TensorRT Optimization)
  \item \textbf{Latency:} $<200\,\text{ms}$ ต่อการ Inference
  \item \textbf{Models:} รัน Ensemble Models พร้อมกัน (ตรวจรอยโรค + ตรวจคุณภาพภาพ)
\end{itemize}

\subsection{Fallback Path}

\begin{itemize}[leftmargin=1.8em, itemsep=3pt]
  \item \textbf{Engine:} TensorFlow Lite (CPU, HTTP via FastAPI)
  \item \textbf{Auto Failover:} $<$1 วินาที หาก Primary Path ล้มเหลว
\end{itemize}

\subsection{ผลลัพธ์เชิงสถาปัตยกรรม}

\begin{tcolorbox}[graybox]
\begin{itemize}[leftmargin=1.4em, itemsep=6pt]
  \item \textbf{Zero-Downtime AI Capability} ---
        สลับจาก Triton (GPU) ไปยัง TFLite (CPU) โดยไม่หยุดให้บริการ
        แม้ AI Server หลักจะ Crash
  \item \textbf{ขจัดปัญหา API Timeout} ---
        AI Inference ประมวลผลแบบ Async (Batch/Queue)
        แก้ปัญหา Request บล็อกภายใต้โหลดสูง
  \item \textbf{Healthcare Compliance Ready} ---
        RBAC และ 2-Tier Audit Logging ตรวจสอบข้อมูลย้อนหลังได้ 100\%
        ตามมาตรฐาน PDPA/HIPAA
  \item \textbf{Modular Design และ Scalability} ---
        Scale Frontend, Backend และ AI ได้อย่างอิสระ
        เพิ่มฟีเจอร์ใหม่โดยไม่กระทบ Component อื่น
  \item \textbf{High Maintainability} ---
        Schema Validation ด้วย Zod + Pydantic
        ลด Technical Debt แบบ End-to-End
\end{itemize}
\end{tcolorbox}
